1"""Project locations stored as GeoJSON geometries (no PostGIS needed). 

2 

3A location is either a Point (a precise spot) or a Polygon (an area, e.g. a 

4street or a park), in WGS84 longitude/latitude as GeoJSON mandates. 

5"""

8from typing import TYPE_CHECKING, Any

10from django.core.exceptions import ValidationError 

11from django.templatetags.static import static 

12from django.utils.translation import gettext_lazy as _ 

13 

14if TYPE_CHECKING: 

15 from publications.models import ProjectPage

17ALLOWED_GEOMETRY_TYPES = ("Point", "Polygon") 

18 

19# Boundary drawn on the public map (Marseille 7e arrondissement, INSEE 13207, 

20# from geo.api.gouv.fr). 

21LOCAL_AREA_BOUNDARY_PATH = "publications/geo/marseille-7e.geojson" 

22# Used before the boundary has loaded and as the admin widget's default view. 

23LOCAL_AREA_CENTER = (43.2805, 5.3530) # lat, lon 

24LOCAL_AREA_ZOOM = 14

27def _is_position(value: Any) -> bool: 

28 return ( 

29 isinstance(value, list) 

30 and len(value) in (2, 3) 

31 and all(isinstance(c, int | float) and not isinstance(c, bool) for c in value) 

32 and -180 <= value[0] <= 180 

33 and -90 <= value[1] <= 90 

34 ) 

35 

36 

37def _is_linear_ring(ring: Any) -> bool: 

38 return ( 

39 isinstance(ring, list) 

40 and len(ring) >= 4 

41 and all(_is_position(p) for p in ring) 

42 and ring[0][:2] == ring[-1][:2] 

43 ) 

44

46def validate_geometry(value: Any) -> None: 

47 """Accept a GeoJSON Point or Polygon geometry, reject anything else.""" 

48 if value in (None, ""): 

49 return 

50 if not isinstance(value, dict) or value.get("type") not in ALLOWED_GEOMETRY_TYPES: 

51 raise ValidationError(_("The location must be a point or a polygon.")) 

52 

53 coordinates = value.get("coordinates") 

54 if value["type"] == "Point": 

55 valid = _is_position(coordinates) 

56 else: 

57 valid = ( 

58 isinstance(coordinates, list) 

59 and len(coordinates) >= 1 

60 and all(_is_linear_ring(ring) for ring in coordinates) 

61 ) 

62 if not valid: 

63 raise ValidationError(_("The location coordinates are invalid."))

66def project_feature(project: "ProjectPage") -> dict[str, Any]: 

67 """GeoJSON Feature describing a project for the public maps.""" 

68 return { 

69 "type": "Feature", 

70 "geometry": project.location, 

71 "properties": { 

72 "title": project.title, 

73 "url": project.url, 

74 "category": project.get_category_display(), 

75 "description": project.description[:200], 

76 "participation": project.participation_mode, 

77 "isOpen": project.is_voting_open or project.is_ideas_open, 

78 }, 

79 } 

80 

81 

82def feature_collection(projects: Iterable["ProjectPage"]) -> dict[str, Any]: 

83 return { 

84 "type": "FeatureCollection", 

85 "features": [project_feature(p) for p in projects if p.location], 

86 }
