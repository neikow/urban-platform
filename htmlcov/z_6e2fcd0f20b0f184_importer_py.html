50def import_archive( 

51 path: str | Path | IO[bytes], mode: str = ImportMode.REPLACE, key: str = "", log: Log = print 

52) -> dict[str, int]: 

53 """Import the archive at ``path`` in one transaction. Raises ArchiveError. 

54 

55 ``replace``: media keep their ids and are updated; pages with the same slug 

56 under the same index page are updated, the home page too. 

57 ``create``: nothing that exists changes. A media id taken by another file 

58 gets a new media (references follow); every page is created, with a free 

59 slug; the home page, one per site, is left as it is. 

60 """ 

61 archive, manifest = open_archive(path, key) 

62 with archive: 

63 importer = _Importer(archive, ImportMode(mode), log) 

64 with transaction.atomic(): 

65 importer.import_media(manifest["images"], manifest["documents"]) 

66 importer.import_pages(manifest["pages"]) 

67 _reset_media_sequences() 

68 return importer.counts
